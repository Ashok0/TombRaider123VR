	uniform sampler2DArray sTex0_mirror;
	in vec4 vTexCoord;
	in vec4 vColor;
	in vec3 vLight;
	in float vLayer;
	in float vFog;
	in vec3 vNormal;
	in vec3 vPos;
	in vec4 vSProj;

	d.xyz *= vLight;
    d.xyz = pow(d.xyz, vec3(2.2, 2.2, 2.2));
	fragColor = d;
}

    uniform vec4 uAmbient[6];
float GetShade(vec3 N, vec3 L)
{
    float NdotL = dot(N, L);
    return max(0.0, NdotL);
}
float GetLight(int index, vec3 pos, vec3 normal)
{
    vec3 lv = uLightPos[index].xyz - pos;
    float att = ((1024.0 / 6.0) * (1024.0 / 6.0)) / dot(lv, lv);
    lv = normalize(lv);
    float NdotL = GetShade(normal, lv);
    return att * NdotL;
}
const vec2 POISSON_DISC[8] = vec2[8](
    (2.0 / 1024.0) * vec2( 0.000000f, 0.000000f),
    (2.0 / 1024.0) * vec2( 0.527837f,-0.085868f),
    (2.0 / 1024.0) * vec2(-0.040088f, 0.536087f),
    (2.0 / 1024.0) * vec2(-0.670445f,-0.179949f),
    (2.0 / 1024.0) * vec2(-0.419418f,-0.616039f),
    (2.0 / 1024.0) * vec2( 0.440453f,-0.639399f),
    (2.0 / 1024.0) * vec2(-0.757088f, 0.349334f),
    (2.0 / 1024.0) * vec2( 0.574619f, 0.685879f)
);
float getShadow(vec4 p, float index)
{
	float s = 0.0;
    p.z = clamp(p.z, 0.0, 1.0);
	for (int i = 0; i < 8; i++)
	{
		vec2 d = POISSON_DISC[i];
		d.xy += p.xy;
		s += texture(sShadow, vec4(vec3(d.x, d.y, p.z).x, vec3(d.x, d.y, p.z).y, index, vec3(d.x, d.y, p.z).z));
	}
    s /= 8;
	return s;
}
float GetPenumbra(vec3 lightVec)
{
    return smoothstep(0.25, 1.0, max(0.0, -lightVec.y));
}
float GetItemShadowLight(int index, vec3 pos, vec3 normal)
{
    vec4 p = ((uShadowMatrix) * (vec4(pos + normal * 16.0, 1.0)));
    p.xyz /= p.w;
    float shadow = getShadow(p, 0);
    vec3 lv = uLightPos[index].xyz - pos * uLightPos[index].w;
    float att = mix(1.0, ((1024.0 / 6.0) * (1024.0 / 6.0)) / dot(lv, lv), uLightPos[index].w);
    lv = normalize(lv);
    float NdotL = GetShade(normal, lv);
    shadow = mix(1.0, shadow, GetPenumbra(lv));
    return att * NdotL * shadow;
}
float GetRoomShadowLight(int index, vec3 pos)
{
    vec4 p = ((uShadowMatrix) * (vec4(pos, 1.0)));
    p.xyz /= p.w;
    float shadow = getShadow(p, 0);
    vec3 lv = uLightPos[index].xyz - pos * uLightPos[index].w;
    lv = normalize(lv);
    return mix(1.0, shadow, GetPenumbra(lv));
}
vec3 getAmbient(vec3 n)
{
    vec3 sqr = n * n;
    vec3 pos = step(vec3(0.0, 0.0, 0.0), n);
    return sqr.x * mix(uAmbient[1].xyz, uAmbient[0].xyz, pos.x) +
           sqr.y * mix(uAmbient[3].xyz, uAmbient[2].xyz, pos.y) +
           sqr.z * mix(uAmbient[5].xyz, uAmbient[4].xyz, pos.z);
}
out vec4 fragColor; void main() {
	vec2 uv = vTexCoord.xy / vTexCoord.zw;
	vec4 d = texture(sTex0, vec3(uv.xy, vLayer)) * vColor;
    vec3 pos = vPos;
    vec3 normal = normalize(vNormal);
    vec3 l = getAmbient(normal);
    l += uLightCol[0].xyz * GetItemShadowLight(0, pos, normal);
    for (int i = 1; i < 4; i++)
    {
        l += uLightCol[i].xyz * GetLight(i, pos, normal);
    }
    d.xyz *= (pow(max((l), vec3(0.0, 0.0, 0.0)), uBrightness.xxx) + (l) * (1.0 - uBrightness.x) * 2.0) * uParams.w; 
    d.xyz = mix(uFogColor.xyz * d.w, d.xyz, vFog);
	fragColor = d;
	if (fragColor.a <= 0.0) discard;
}

vec4 shadeUW(vec4 d, vec3 p, vec3 wp)
{
    float h = wp.y - uModelMatrix[3].y;
    float k = texture(sNoise, wp.xzy / (16.0 * 1024.0) - vec3(0.0, 0.0, uModelMatrix[3].x * 0.1)).x;
    d.xyz *= 0.25 + k * 2.0;
    float dist = length(p);
    if (vec3(uViewMatrix[0].w, uViewMatrix[1].w, uViewMatrix[2].w).y < uModelMatrix[3].y)
    {
        dist *= h / p.y;
    }
    dist += min(h, 2048.0);
    d.xyz *= mix(vec3(1.0, 1.0, 1.0), vec3(0.6, 0.843, 0.901), clamp(sqrt(dist * (1.0 / 2048.0)), 0.0, 2.0));
    d.xyz = mix(vec3(0.0036, 0.0169, 0.0231) * d.w, d.xyz, clamp(exp(-((dist / (12.0 * 1024.0))*(dist / (12.0 * 1024.0)))), 0.0, 1.0));
    return d;
}

    }

{
    float NdotL = dot(N, L);
    return max(0.0, NdotL);
}
float GetLight(int index, vec3 pos, vec3 normal)
{
    vec3 lv = uLightPos[index].xyz - pos;
    float att = ((1024.0 / 6.0) * (1024.0 / 6.0)) / dot(lv, lv);
    lv = normalize(lv);
    float NdotL = GetShade(normal, lv);
    return att * NdotL;

vec4 shadeUW(vec4 d, vec3 p, vec3 wp)
{
    float h = wp.y - uModelMatrix[3].y;
    float k = texture(sNoise, wp.xzy / (16.0 * 1024.0) - vec3(0.0, 0.0, uModelMatrix[3].x * 0.1)).x;
    d.xyz *= 0.25 + k * 2.0;
    float dist = length(p);
    if (vec3(uViewMatrix[0].w, uViewMatrix[1].w, uViewMatrix[2].w).y < uModelMatrix[3].y)
    {
        dist *= h / p.y;
    }
    dist += min(h, 2048.0);
    d.xyz *= mix(vec3(1.0, 1.0, 1.0), vec3(0.6, 0.843, 0.901), clamp(sqrt(dist * (1.0 / 2048.0)), 0.0, 2.0));
    d.xyz = mix(vec3(0.0036, 0.0169, 0.0231), d.xyz, clamp(exp(-((dist / (12.0 * 1024.0))*(dist / (12.0 * 1024.0)))), 0.0, 1.0));
    return d;
}

vec2 oct3to2(vec3 p)
{
	p /= abs(p.x) + abs(p.y) + abs(p.z);
    if (p.y < 0.0)
		p.xz = (1.0 - abs(p.zx)) * sign(p.xz);
	return p.xz * 0.5 + 0.5;
}
vec3 fresnel(vec3 diffuse, vec3 viewVec, vec3 normal, float metalness)
{
    const float F0 = 0.04;
    float NdotV = dot(normal, viewVec);
    float Fd = F0 + (1.0 - F0) * pow(1.0 - NdotV, 5.0);
    vec3 Fm = mix(vec3(F0, F0, F0), diffuse, metalness);
    return mix(vec3(Fd, Fd, Fd), Fm, metalness);
}

    uv += uParams.xy * uModelMatrix[3].x;

    vColor = pow(color.xyz, vec3(2.2, 2.2, 2.2));
    vLayer = normal.w;

    vec3 wp = p.xyz + vec3(uViewMatrix[0].w, uViewMatrix[1].w, uViewMatrix[2].w);
    float weight = coord.w / 32767.0;
    float k = texture(sNoise, wp * uParams.z).x;
    float t = uModelMatrix[3].x * 1.25;
    float offset = sin(k * (3.1415926536 * 2.0) + t) * weight * uParams.w;

vec2 oct3to2(vec3 p)
{
	p /= abs(p.x) + abs(p.y) + abs(p.z);
    if (p.y < 0.0)
		p.xz = (1.0 - abs(p.zx)) * sign(p.xz);
	return p.xz * 0.5 + 0.5;
}
vec3 fresnel(vec3 diffuse, vec3 viewVec, vec3 normal, float metalness)
{
    const float F0 = 0.04;
    float NdotV = dot(normal, viewVec);
    float Fd = F0 + (1.0 - F0) * pow(1.0 - NdotV, 5.0);
    vec3 Fm = mix(vec3(F0, F0, F0), diffuse, metalness);
    return mix(vec3(Fd, Fd, Fd), Fm, metalness);
}

    }
    d.xyz *= (pow(max((light), vec3(0.0, 0.0, 0.0)), uBrightness.xxx) + (light) * (1.0 - uBrightness.x) * 2.0) * 2.0;

    vec3 oct = texture(sTex0_mirror, vec3(oct_uv.xy, uParams.x)).xyz;
        d.xyz *= 1.0 - d.w;

float GetShade(vec3 N, vec3 L)
{
    float NdotL = dot(N, L);
    return max(0.0, NdotL);
}
float GetLight(int index, vec3 pos, vec3 normal)
{
    vec3 lv = uLightPos[index].xyz - pos;
    float att = ((1024.0 / 6.0) * (1024.0 / 6.0)) / dot(lv, lv);
    lv = normalize(lv);
    float NdotL = GetShade(normal, lv);
    return att * NdotL;
}

float getShadow(vec4 p, float index)
{
	float s = 0.0;
    p.z = clamp(p.z, 0.0, 1.0);
	for (int i = 0; i < 8; i++)
	{
		vec2 d = POISSON_DISC[i];
		d.xy += p.xy;
		s += texture(sShadow, vec4(vec3(d.x, d.y, p.z).x, vec3(d.x, d.y, p.z).y, index, vec3(d.x, d.y, p.z).z));
	}
    s /= 8;
	return s;
}
float GetPenumbra(vec3 lightVec)
{
    return smoothstep(0.25, 1.0, max(0.0, -lightVec.y));
}
float GetItemShadowLight(int index, vec3 pos, vec3 normal)
{
    vec4 p = ((uShadowMatrix) * (vec4(pos + normal * 16.0, 1.0)));
    p.xyz /= p.w;
    float shadow = getShadow(p, 0);
    vec3 lv = uLightPos[index].xyz - pos * uLightPos[index].w;
    float att = mix(1.0, ((1024.0 / 6.0) * (1024.0 / 6.0)) / dot(lv, lv), uLightPos[index].w);
    lv = normalize(lv);
    float NdotL = GetShade(normal, lv);
    shadow = mix(1.0, shadow, GetPenumbra(lv));
    return att * NdotL * shadow;
}
float GetRoomShadowLight(int index, vec3 pos)
{
    vec4 p = ((uShadowMatrix) * (vec4(pos, 1.0)));
    p.xyz /= p.w;
    float shadow = getShadow(p, 0);
    vec3 lv = uLightPos[index].xyz - pos * uLightPos[index].w;
    lv = normalize(lv);
    return mix(1.0, shadow, GetPenumbra(lv));
}

vec2 oct3to2(vec3 p)
{
	p /= abs(p.x) + abs(p.y) + abs(p.z);
    if (p.y < 0.0)
		p.xz = (1.0 - abs(p.zx)) * sign(p.xz);
	return p.xz * 0.5 + 0.5;
}
vec3 fresnel(vec3 diffuse, vec3 viewVec, vec3 normal, float metalness)
{
    const float F0 = 0.04;
    float NdotV = dot(normal, viewVec);
    float Fd = F0 + (1.0 - F0) * pow(1.0 - NdotV, 5.0);
    vec3 Fm = mix(vec3(F0, F0, F0), diffuse, metalness);
    return mix(vec3(Fd, Fd, Fd), Fm, metalness);
}

    float hX = texture(sNoise, vec3(uv + d.xy, t)).x;
    float hY = texture(sNoise, vec3(uv + d.yx, t)).x;
    hX -= hC;
    hY -= hC;
    vec3 n = cross(vec3(0.0, hY * (1.0 / 16.0), (1.0 / 128.0)), vec3((1.0 / 128.0), hX * (1.0 / 16.0), 0.0));
    n = -normalize(n);
    if (!gl_FrontFacing)
    {
        n = -n;
    }
    vec3 pos = vPos;
    vec3 vv = normalize(-pos);
    float NdotV = max(0.0, dot(n, -vv));
    vec3 r = normalize(reflect(-vv, n));
    vec3 sqr = r * r;
    vec3 p = step(vec3(0.0, 0.0, 0.0), r);
    vec3 ref =
           sqr.x * mix(vrgb1, vrgb0, p.x) +
           sqr.y * mix(vrgb3, vrgb2, p.y) +
           sqr.z * mix(vrgb5, vrgb4, p.z);
    float F0 = 0.04;
    float Fd = F0 + (1.0 - F0) * pow(1.0 - NdotV, 3.0);
    vec4 color = vec4(ref * Fd * 2.0, 0.0);
    fragColor = color;
}
